# 日志系统回溯能力增强方案

## 回溯需求分析

基于用户对"代码架构回溯"和"代码失败无死角回溯"的需求，当前整改方案需要进一步增强：

### 核心回溯需求
1. **代码架构回溯**
   - 追踪模块使用变化和架构偏离
   - 记录架构决策和变更原因
   - 检测代码重复和轮子重造

2. **代码失败无死角回溯**
   - 捕获所有错误类型（同步、异步、未处理）
   - 记录完整的错误上下文和调用链
   - 提供时间线重建能力

### 1. 架构变更追踪系统

```javascript
// src/core/ArchitectureChangeTracker.js
class ArchitectureChangeTracker {
    constructor() {
        this.changeLog = [];
        this.moduleRegistry = new Map();
        this.archRules = this._loadArchitectureRules();
    }

    /**
     * 记录模块使用变化
     */
    trackModuleUsage(moduleName, operation, context) {
        const change = {
            timestamp: new Date().toISOString(),
            module: moduleName,
            operation, // 'create', 'modify', 'delete', 'usage'
            context,
            stack: new Error().stack,
            hash: this._computeCodeHash(moduleName)
        };

        this.changeLog.push(change);
        
        // 检查架构合规性
        const violation = this._checkArchitectureCompliance(change);
        if (violation) {
            this._recordArchitectureViolation(violation);
        }

        // 持久化架构变更
        this._persistArchitectureChange(change);
    }

    /**
     * 检测代码重复
     */
    detectCodeDuplication() {
        const moduleHashes = new Map();
        const duplicates = [];

        this.moduleRegistry.forEach((module, name) => {
            const hash = this._computeCodeHash(module);
            if (moduleHashes.has(hash)) {
                duplicates.push({
                    module1: moduleHashes.get(hash),
                    module2: name,
                    hash: hash,
                    timestamp: new Date().toISOString()
                });
            } else {
                moduleHashes.set(hash, name);
            }
        });

        return duplicates;
    }

    /**
     * 架构回溯查询
     */
    queryArchitectureHistory(filters) {
        return this.changeLog.filter(change => {
            if (filters.module && change.module !== filters.module) return false;
            if (filters.operation && change.operation !== filters.operation) return false;
            if (filters.since && new Date(change.timestamp) < new Date(filters.since)) return false;
            return true;
        }).sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));
    }
}
```

### 2. 无死角错误回溯系统

```javascript
// src/core/ComprehensiveErrorTracker.js
class ComprehensiveErrorTracker {
    constructor() {
        this.errorLog = [];
        this.contextStack = [];
        this._setupGlobalErrorHandlers();
    }

    /**
     * 设置全局错误处理器
     */
    _setupGlobalErrorHandlers() {
        // 1. 同步错误捕获
        window.addEventListener('error', (event) => {
            this._handleError('SYNC_ERROR', event.error, {
                filename: event.filename,
                lineno: event.lineno,
                colno: event.colno
            });
        });

        // 2. 异步错误捕获
        window.addEventListener('unhandledrejection', (event) => {
            this._handleError('ASYNC_ERROR', event.reason, {
                type: 'UnhandledPromiseRejection'
            });
        });

        // 3. 控制台错误重定向
        const originalConsoleError = console.error;
        console.error = (...args) => {
            this._handleError('CONSOLE_ERROR', args.join(' '), {
                args: args,
                stack: new Error().stack
            });
            originalConsoleError.apply(console, args);
        };

        // 4. 网络错误监控
        this._interceptFetchErrors();
        
        // 5. 资源加载错误
        window.addEventListener('load', () => {
            window.addEventListener('error', (e) => {
                if (e.target && e.target.src) {
                    this._handleError('RESOURCE_ERROR', `资源加载失败: ${e.target.src}`, {
                        element: e.target.tagName,
                        src: e.target.src
                    });
                }
            }, true);
        });
    }

    /**
     * 统一错误处理
     */
    _handleError(type, error, context) {
        const errorEntry = {
            id: this._generateErrorId(),
            timestamp: new Date().toISOString(),
            type: type,
            message: error instanceof Error ? error.message : String(error),
            stack: error instanceof Error ? error.stack : new Error().stack,
            context: {
                ...context,
                currentUrl: window.location.href,
                userAgent: navigator.userAgent,
                contextStack: [...this.contextStack] // 当前上下文链
            },
            severity: this._calculateErrorSeverity(type, error)
        };

        // 记录到错误日志
        this.errorLog.push(errorEntry);

        // 持久化重要错误
        if (errorEntry.severity >= this._getConfig().minPersistSeverity) {
            this._persistError(errorEntry);
        }

        // 触发错误事件
        this._emitErrorEvent(errorEntry);
    }

    /**
     * 上下文追踪
     */
    pushContext(context) {
        this.contextStack.push({
            ...context,
            timestamp: new Date().toISOString(),
            stack: new Error().stack
        });
        
        // 保持上下文栈大小
        if (this.contextStack.length > 20) {
            this.contextStack.shift();
        }
    }

    popContext() {
        return this.contextStack.pop();
    }

    /**
     * 错误回溯查询
     */
    queryErrorHistory(filters) {
        let results = this.errorLog;

        if (filters.type) {
            results = results.filter(error => error.type === filters.type);
        }
        if (filters.severity) {
            results = results.filter(error => error.severity >= filters.severity);
        }
        if (filters.since) {
            results = results.filter(error => new Date(error.timestamp) >= new Date(filters.since));
        }
        if (filters.search) {
            const search = filters.search.toLowerCase();
            results = results.filter(error => 
                error.message.toLowerCase().includes(search) ||
                error.stack.toLowerCase().includes(search)
            );
        }

        return results.sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));
    }

    /**
     * 时间线重建
     */
    reconstructTimeline(startTime, endTime) {
        const timeline = [];
        
        // 合并架构变更和错误日志
        const architectureChanges = window.ArchitectureChangeTracker?.queryArchitectureHistory({
            since: startTime
        }) || [];

        const errors = this.queryErrorHistory({
            since: startTime
        });

        // 按时间排序
        const allEvents = [...architectureChanges, ...errors].sort((a, b) => 
            new Date(a.timestamp) - new Date(b.timestamp)
        );

        // 构建因果关系
        return this._buildCausalRelationships(allEvents);
    }
}
```

```javascript
// src/core/EnhancedLogContext.js
class EnhancedLogContext {
    constructor() {
        this.sessionId = this._generateSessionId();
        this.userJourney = [];
        this.performanceMarkers = [];
    }

    /**
     * 记录用户旅程
     */
    recordUserAction(action, data) {
        const journeyStep = {
            action,
            data,
            timestamp: new Date().toISOString(),
            url: window.location.href,
            stack: new Error().stack,
            performance: performance.now()
        };

        this.userJourney.push(journeyStep);
        
        // 保持旅程记录大小
        if (this.userJourney.length > 100) {
            this.userJourney.shift();
        }

        // 添加到日志上下文
        window.UnifiedLoggingFacade?.pushContext({
            type: 'USER_ACTION',
            action: action,
            journeyStep: journeyStep
        });
    }

    /**
     * 性能标记
     */
    markPerformance(marker, data) {
        this.performanceMarkers.push({
            marker,
            data,
            timestamp: new Date().toISOString(),
            performance: performance.now()
        });
    }

    /**
     * 获取完整上下文
     */
    getFullContext() {
        return {
            sessionId: this.sessionId,
            userJourney: [...this.userJourney],
            performanceMarkers: [...this.performanceMarkers],
            currentUrl: window.location.href,
            userAgent: navigator.userAgent,
            timestamp: new Date().toISOString()
        };
    }
}
```

```javascript
class BacktraceQueryPanel {

                    <h4>时间线重建</h4>
                    <input type="datetime-local" id="timeline-start">
                    <input type="datetime-local" id="timeline-end">
                    <button onclick="this.reconstructTimeline()">重建时间线</button>
                </div>

                <div class="results-section" id="backtrace-results"></div>
            </div>
        `;
    }

    /**
     * 查询错误历史
     */
    async queryErrors() {
        const filters = {
            type: document.getElementById('error-type-filter').value,
            since: document.getElementById('error-time-start').value,
            until: document.getElementById('error-time-end').value
        };

        const results = await this.tracker.queryErrorHistory(filters);
        this._displayResults('errors', results);
    }

    /**
     * 查询架构变更
     */
    async queryArchChanges() {
        const filters = {
            module: document.getElementById('module-filter').value,
            operation: document.getElementById('operation-filter').value
        };

        const results = await this.archTracker.queryArchitectureHistory(filters);
        this._displayResults('architecture', results);
    }

    /**
     * 重建时间线
     */
    async reconstructTimeline() {
        const start = document.getElementById('timeline-start').value;
        const end = document.getElementById('timeline-end').value;
        
        const timeline = await this.tracker.reconstructTimeline(start, end);
        this._displayResults('timeline', timeline);
    }
}
```

### 修改统一日志门面

```javascript
// 更新 UnifiedLoggingFacade
class UnifiedLoggingFacade {
    constructor() {
        // 现有组件
        this.logger = null;
        this.collector = null;
        this.panel = null;
        this.storage = null;
        
        // 新增回溯组件
        this.archTracker = new ArchitectureChangeTracker();
        this.errorTracker = new ComprehensiveErrorTracker();
        this.logContext = new EnhancedLogContext();
        
        this._initialize();
    }

    /**
     * 增强的日志方法
     */
    log(level, category, message, data, context) {
        const enhancedContext = {
            ...this.logContext.getFullContext(),
            ...context
        };

        // 记录到架构追踪器（如果是架构相关）
        if (category === 'architecture' || level === 'WARN' || level === 'ERROR') {
            this.archTracker.trackModuleUsage(category, 'log', {
                level, message, data, context: enhancedContext
            });
        }

        // 记录到错误追踪器（如果是错误）
        if (level === 'ERROR') {
            this.errorTracker.pushContext(enhancedContext);
        }

        // 原有日志逻辑
        this.logger.log(level, category, message, data, enhancedContext);
        this.collector.addLog(level, message, data);
        this.panel.append(level, message);
        this._persistAsync(level, category, message, data, enhancedContext);
    }
}
```
