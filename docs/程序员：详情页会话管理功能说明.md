### **2. 会话解析**
```javascript
_parseSessions(text) {
    const sessions = [];
    const lines = text.split('\n');
    let current = null;
    
    lines.forEach((line, idx) => {
        const match = line.match(/^##\s+(.+)$/);
        if (match) {
            if (current) sessions.push(current);
            current = {
                title: match[1],
                startLine: idx,
                raw: ''
            };
        } else if (current) {
            current.raw += line + '\n';
        }
    });
    
    if (current) sessions.push(current);
    return sessions;
}
```

### **3. 会话渲染**
```javascript
_renderSessionList() {
    const itemsBox = this._mmFsOverlay.querySelector('.mm-session-items');
    const sessions = this._parseSessions(this._mmFsTextarea.value);
    
    itemsBox.innerHTML = '';
    sessions.forEach((s, idx) => {
        const item = document.createElement('div');
        item.className = 'mm-session-item';
        item.innerHTML = `
            <label class="mm-session-check">
                <input type="checkbox" class="mm-session-check-cb" />
            </label>
            <span class="mm-session-title">${s.title}</span>
        `;
        
        // 点击定位
        item.addEventListener('click', () => this._gotoSession(idx));
        
        // 双击重命名
        item.addEventListener('dblclick', () => {
            const newTitle = prompt('重命名会话', s.title);
            if (newTitle) this._replaceSessionTitle(idx, newTitle);
        });
        
        itemsBox.appendChild(item);
    });
}
```

### **4. 定位会话**
```javascript
_gotoSession(index) {
    const sessions = this._parseSessions(this._mmFsTextarea.value);
    const session = sessions[index];
    if (!session) return;
    
    const lines = this._mmFsTextarea.value.split('\n');
    const lineHeight = parseInt(window.getComputedStyle(this._mmFsTextarea).lineHeight);
    const scrollTop = session.startLine * lineHeight;
    
    this._mmFsTextarea.scrollTop = scrollTop;
    this._mmFsTextarea.focus();
}
```

### **5. 新增会话**
```javascript
_addSessionAtCursor(textarea) {
    const pos = textarea.selectionStart;
    const before = textarea.value.substring(0, pos);
    const after = textarea.value.substring(pos);
    
    const newSession = '\n\n## 新会话\n\n';
    textarea.value = before + newSession + after;
    
    // 更新光标位置
    textarea.selectionStart = pos + newSession.length - 1;
    textarea.selectionEnd = pos + newSession.length - 1;
    textarea.focus();
    
    // 重新渲染会话列表
    this._renderSessionList();
}
```
