// 拖拽多屏管理器
class DragMultiScreenManager {
    private dragInstances: Map<string, DragInstance> = new Map();
    
    constructor() {
        this.setupDragAndDrop();
        this.setupScreenDetection();
    }
    
    // 设置拖拽功能
    setupDragAndDrop(): void {
        // 使用成熟的拖拽库 (如 interact.js)
        interact('.column-app')
            .draggable({
                listeners: {
                    start: (event) => this.handleDragStart(event),
                    move: (event) => this.handleDragMove(event),
                    end: (event) => this.handleDragEnd(event)
                }
            })
            .resizable({
                edges: { left: true, right: true, bottom: true, top: true },
                listeners: {
                    move: (event) => this.handleResize(event)
                }
            });
    }
    
    // 处理跨屏幕拖拽
    handleDragEnd(event: DragEvent): void {
        const appElement = event.target as HTMLElement;
        const appId = appElement.dataset.appId;
        
        // 检测是否拖拽到其他屏幕
        const screenBounds = this.getScreenBounds();
        const elementBounds = appElement.getBoundingClientRect();
        
        screenBounds.forEach((bounds, screenId) => {
            if (this.isElementInScreen(elementBounds, bounds)) {
                this.moveAppToScreen(appId, screenId);
            }
        });
    }
    
    // 移动应用到指定屏幕
    moveAppToScreen(appId: string, screenId: string): void {
        const app = this.appInstances.get(appId);
        if (app && screenId !== 'primary') {
            // 创建新窗口到目标屏幕
            const newWindow = this.createWindowOnScreen(screenId);
            app.moveToWindow(newWindow);
        }
    }
}

                <div class="divider-handle"></div>
            </div>
        `;
        
        // 根据position插入到正确位置
        const [pos, targetId] = position.split(':');
        const targetColumn = document.getElementById(`${targetId}-column`);
        
        if (pos === 'after' && targetColumn) {
            targetColumn.insertAdjacentHTML('afterend', html);
        }
    }
    
    _injectToggleButton(id, title, icon) {
        const btn = document.createElement('button');
        btn.className = 'view-toggle';
        btn.dataset.view = id;
        btn.textContent = icon ? `${icon} ${title}` : title;
        
        // 插入到.view-toggles容器
        const container = document.querySelector('.view-toggles');
        if (container) {
            container.appendChild(btn);
        }
    }
    
    /**
     * 移除工作栏 - 一行代码完全清理
     */
    unregister(id) {
        // 1. 移除HTML
        document.getElementById(`${id}-column`)?.remove();
        document.getElementById(`${id}-divider`)?.remove();
        
        // 2. 移除按钮
        document.querySelector(`.view-toggle[data-view="${id}"]`)?.remove();
        
        // 3. 从ColumnManager移除
        if (window.columnManager) {
            window.columnManager.views = window.columnManager.views.filter(v => v !== id);
            window.columnManager.activeViews = window.columnManager.activeViews.filter(v => v !== id);
        }
        
        // 4. 清理存储
        this.columns.delete(id);
        
        // 5. 触发事件
        window.AutogenEventBus.emit('column:unregistered', { id });
        
        console.log(`✅ 工作栏 "${id}" 已完全移除`);
    }
}

// 全局实例
window.ColumnRegistry = new ColumnRegistry();

    constructor() {
        // 启动时自动恢复
        this._restoreUserColumns();
    }
    
    register(config) {
        // ... 注册逻辑
        
        // 自动保存到localStorage
        this._saveToStorage(config);
    }
    
    _saveToStorage(config) {
        const saved = JSON.parse(localStorage.getItem('user-columns') || '[]');
        saved.push({
            id: config.id,
            title: config.title,
            type: config.type,
            position: config.position
        });
        localStorage.setItem('user-columns', JSON.stringify(saved));
    }
    
    _restoreUserColumns() {
        const saved = JSON.parse(localStorage.getItem('user-columns') || '[]');
        saved.forEach(config => {
            this.register(config);
        });
    }
}

                                🗑️
                            </button>

    _initDragSort() {
        const list = document.getElementById('column-order-list');
        let draggedItem = null;
        
        list.addEventListener('dragstart', (e) => {
            if (e.target.classList.contains('column-item')) {
                draggedItem = e.target;
                e.target.style.opacity = '0.5';
            }
        });
        
        list.addEventListener('dragend', (e) => {
            if (e.target.classList.contains('column-item')) {
                e.target.style.opacity = '1';
            }
        });
        
        list.addEventListener('dragover', (e) => {
            e.preventDefault();
            const afterElement = this._getDragAfterElement(list, e.clientY);
            if (afterElement == null) {
                list.appendChild(draggedItem);
            } else {
                list.insertBefore(draggedItem, afterElement);
            }
        });
    }
    
    _getDragAfterElement(container, y) {
        const draggableElements = [...container.querySelectorAll('.column-item:not(.dragging)')];
        
        return draggableElements.reduce((closest, child) => {
            const box = child.getBoundingClientRect();
            const offset = y - box.top - box.height / 2;
            
            if (offset < 0 && offset > closest.offset) {
                return { offset: offset, element: child };
            } else {
                return closest;
            }
        }, { offset: Number.NEGATIVE_INFINITY }).element;
    }

    async _importColumnFile(file) {
        return new Promise((resolve, reject) => {
            const reader = new FileReader();
            
            reader.onload = (e) => {
                try {
                    const code = e.target.result;
                    
                    // 解析工作栏元数据
                    const metadata = this._parseColumnMetadata(code);
                    
                    // 存入仓库
                    this.warehouse.set(metadata.id, {
                        ...metadata,
                        code: code,
                        fileName: file.name,
                        uploadTime: new Date().toISOString(),
                        isOnline: false
                    });
                    
                    // 保存到localStorage
                    this._saveWarehouse();
                    
                    console.log(`[ColumnWarehouse] ✅ 入库成功: ${metadata.name}`);
                    resolve(metadata);
                    
                } catch (err) {
                    reject(err);
                }
            };
            
            reader.onerror = () => reject(new Error('文件读取失败'));
            reader.readAsText(file);
        });
    }

        try {
            // 使用Function构造器在隔离环境中执行

    _extractMetadataFromCode(code) {
        const idMatch = code.match(/id:\s*['"]([^'"]+)['"]/);
        const nameMatch = code.match(/name:\s*['"]([^'"]+)['"]/);
        const versionMatch = code.match(/version:\s*['"]([^'"]+)['"]/);
        const iconMatch = code.match(/icon:\s*['"]([^'"]+)['"]/);
        
        return {
            id: idMatch ? idMatch[1] : 'unknown-' + Date.now(),
            name: nameMatch ? nameMatch[1] : '未命名工作栏',
            version: versionMatch ? versionMatch[1] : '1.0.0',
            icon: iconMatch ? iconMatch[1] : '📄'
        };
