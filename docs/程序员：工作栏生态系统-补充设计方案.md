## 1️⃣ **跨数据类型统一打标签机制**

### **1.1 核心设计**

#### **基于焦点的标签应用**

```javascript
// 标签管理栏核心逻辑
class TagManagerColumn {
    /**
     * 应用标签到当前焦点项
     */
    async applyTag(tagId) {
        // 获取当前焦点数据
        const currentData = this.getCurrentSelectedData();
        
        if (!currentData) {
            alert('请先选择要打标签的项目');
            return;
        }
        
        // 从数据底座获取数据
        const data = await AutogenUnifiedStorage.retrieve(
            currentData.type, 
            currentData.id
        );
        
        // 添加标签到metadata
        if (!data.metadata) data.metadata = {};
        if (!data.metadata.tags) data.metadata.tags = [];
        
        const tag = await AutogenUnifiedStorage.retrieve('tag', tagId);
        if (!data.metadata.tags.includes(tag.data.name)) {
            data.metadata.tags.push(tag.data.name);
            
            // 更新存储
            await AutogenUnifiedStorage.store(
                currentData.type, 
                currentData.id, 
                data.data, 
                { metadata: data.metadata }
            );
            
            // 触发事件通知其他工作栏
            AutogenEventBus.emit('tag.applied', {
                targetType: currentData.type,
                targetId: currentData.id,
                tagId: tagId,
                tagName: tag.data.name
            });
        }
    }
    
    /**
     * 获取当前选中的数据（简单方式）
     */
    getCurrentSelectedData() {
        // 脑图栏的选中节点
        if (window.mindmapController?.selectedNode) {
            return {
                type: 'mindmap',
                id: window.mindmapController.selectedNode.id
            };
        }
        
        // 列表栏的选中项
        if (window.currentSelectedListItem) {
            return {
                type: 'task',
                id: window.currentSelectedListItem
            };
        }
        
        // 详情栏当前显示的项
        if (window.currentDetailItemId) {
            return {
                type: 'task',
                id: window.currentDetailItemId
            };
        }
        
        return null;
    }
}
```

## 2️⃣ **实时数据同步机制**

### **2.1 核心架构**

```
AutogenUnifiedStorage (数据底座)
        ↓ 数据变更
AutogenEventBus (事件通知)
        ↓ 广播事件
所有工作栏 (自动监听刷新)
```

### **2.2 数据底座增强**

```javascript
// 在 AutogenUnifiedStorage.store() 方法中新增
async store(type, key, data, options = {}) {
    try {
        // ... 原有存储逻辑 ...
        
        // ✅ 存储成功后，自动触发数据变更事件
        if (global.AutogenEventBus) {
            // 通用事件
            global.AutogenEventBus.emit('storage.dataChanged', {
                type: type,
                key: key,
                action: 'update',
                timestamp: Date.now()
            });
            
            // 类型特定事件（更精确）
            global.AutogenEventBus.emit(`storage.${type}.changed`, {
                key: key,
                action: 'update',
                data: data
            });
        }
        
        return true;
    } catch (error) {
        // ... 错误处理 ...
    }
}
```

### **2.4 性能优化**

#### **防抖处理**
```javascript
class ListColumn {
    constructor() {
        this.refreshDebounce = new Map();
    }
    
    refreshListItem(taskId) {
        // 清除之前的定时器
        if (this.refreshDebounce.has(taskId)) {
            clearTimeout(this.refreshDebounce.get(taskId));
        }
        
        // 300ms后执行
        const timer = setTimeout(async () => {
            await this._doRefreshListItem(taskId);
            this.refreshDebounce.delete(taskId);
        }, 300);
        
        this.refreshDebounce.set(taskId, timer);
    }
}
```

### **3.3 ConfigManager实现**

```javascript
class ConfigManager {
    constructor() {
        this.config = null;
        this.defaultConfig = this.getDefaultConfig();
    }
    
    async init() {
        await this.loadConfig();
    }
    
    async loadConfig() {
        const saved = await AutogenUnifiedStorage.retrieve('app_config', 'main');
        
        if (saved) {
            this.config = saved.data;
        } else {
            this.config = this.defaultConfig;
            await this.saveConfig();
        }
    }
    
    async saveConfig() {
        await AutogenUnifiedStorage.store('app_config', 'main', this.config, {
            metadata: { itemType: 'app_config' }
        });
        
        // 触发配置变更事件
        AutogenEventBus.emit('config.changed', {
            config: this.config
        });
    }
    
    // 获取配置值（支持路径）
    get(path) {
        const keys = path.split('.');
        let value = this.config;
        
        for (const key of keys) {
            value = value?.[key];
            if (value === undefined) return null;
        }
        
        return value;
    }
    
    // 设置配置值
    async set(path, value) {
        const keys = path.split('.');
        const lastKey = keys.pop();
        let target = this.config;
        
        for (const key of keys) {
            if (!target[key]) target[key] = {};
            target = target[key];
        }
        
        target[lastKey] = value;
        await this.saveConfig();
    }
    
    // 获取标签样式
    getTagStyle(tagName) {
        return this.get(`tagStyles.${tagName}`) || {
            color: '#6b7280',
            nodeStyle: {
                borderColor: '#d1d5db',
                borderWidth: 2
            }
        };
    }
    
    // 设置标签样式
    async setTagStyle(tagName, style) {
        await this.set(`tagStyles.${tagName}`, style);
    }
    
    getDefaultConfig() {
        return {
            tagStyles: {
                '重要': {
                    color: '#ef4444',
                    nodeStyle: {
                        borderColor: '#ef4444',
                        borderWidth: 3,
                        fontWeight: 'bold',
                        backgroundColor: '#fee2e2'
                    }
                },
                '紧急': {
                    color: '#f59e0b',
                    nodeStyle: {
                        borderColor: '#f59e0b',
                        borderWidth: 3,
                        fontWeight: 'bold',
                        backgroundColor: '#fef3c7'
                    }
                },
                '一般': {
                    color: '#6b7280',
                    nodeStyle: {
                        borderColor: '#d1d5db',
                        borderWidth: 2
                    }
                }
            },
            columnLayout: {
                mode: 'equal',
                columns: {}
            },
            theme: {
                primaryColor: '#3b82f6',
                fontSize: '14px'
            }
        };
    }
}

window.ConfigManager = new ConfigManager();
```
