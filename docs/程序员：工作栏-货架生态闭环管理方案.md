### **1. ColumnWarehouse 核心类**

```javascript
/**
 * 工作栏仓库管理器
 * 负责工作栏的入库、上线、下线、删除
 */
class ColumnWarehouse {
    constructor() {
        this.warehouse = new Map(); // 库房中的所有工作栏
        this.onlineColumns = [];    // 已上线的工作栏ID列表
        this._init();
    }
    
    _init() {
        // 从localStorage加载仓库数据
        this._loadWarehouse();
        
        // 创建仓库UI（隐藏状态）
        this._createWarehouseUI();
        
        console.log('[ColumnWarehouse] ✅ 初始化完成');
    }
    
    /**
     * 打开仓库管理界面
     */
    open() {
        const dialog = document.getElementById('column-warehouse-dialog');
        dialog.style.display = 'flex';
        
        // 刷新显示
        this._refreshWarehouseList();
        this._refreshOnlineList();
    }
    
    /**
     * 关闭仓库
     */
    close() {
        const dialog = document.getElementById('column-warehouse-dialog');
        dialog.style.display = 'none';
    }
    
    /**
     * 上传工作栏（入库）
     */
    async uploadColumn() {
        // 创建文件选择器
        const input = document.createElement('input');
        input.type = 'file';
        input.accept = '.js';
        input.multiple = true;
        
        input.onchange = async (e) => {
            const files = Array.from(e.target.files);
            
            for (const file of files) {
                try {
                    await this._importColumnFile(file);
                } catch (err) {
                    console.error('[ColumnWarehouse] 上传失败:', err);
                    alert(`上传 ${file.name} 失败: ${err.message}`);
                }
            }
            
            this._refreshWarehouseList();
        };
        
        input.click();
    }
    
    /**
     * 导入工作栏文件
     */
    async _importColumnFile(file) {
        return new Promise((resolve, reject) => {
            const reader = new FileReader();
            
            reader.onload = (e) => {
                try {
                    const code = e.target.result;
                    
                    // 解析工作栏元数据
                    const metadata = this._parseColumnMetadata(code);
                    
                    // 存入仓库
                    this.warehouse.set(metadata.id, {
                        ...metadata,
                        code: code,
                        fileName: file.name,
                        uploadTime: new Date().toISOString(),
                        isOnline: false
                    });
                    
                    // 保存到localStorage
                    this._saveWarehouse();
                    
                    console.log(`[ColumnWarehouse] ✅ 入库成功: ${metadata.name}`);
                    resolve(metadata);
                    
                } catch (err) {
                    reject(err);
                }
            };
            
            reader.onerror = () => reject(new Error('文件读取失败'));
            reader.readAsText(file);
        });
    }
    
    /**
     * 解析工作栏元数据
     */
    _parseColumnMetadata(code) {
        // 临时执行代码提取元数据
        const tempContext = {};
        
        try {
            // 使用Function构造器在隔离环境中执行
            const fn = new Function('context', code + '\nreturn window.lastRegisteredColumn;');
            
            // 创建临时window环境
            const tempWindow = {
                ColumnRegistry: {
                    register: (config) => {
                        tempWindow.lastRegisteredColumn = config;
                    }
                }
            };
            
            // 执行并获取元数据
            const oldWindow = window;
            window.lastRegisteredColumn = null;
            
            eval(code); // 简化版，实际应使用沙箱
            
            const metadata = window.lastRegisteredColumn || this._extractMetadataFromCode(code);
            
            if (!metadata || !metadata.id) {
                throw new Error('无法解析工作栏元数据');
            }
            
            return metadata;
            
        } catch (err) {
            // 回退：使用正则提取
            return this._extractMetadataFromCode(code);
        }
    }
    
    /**
     * 从代码中提取元数据（正则方式）
     */
    _extractMetadataFromCode(code) {
        const idMatch = code.match(/id:\s*['"]([^'"]+)['"]/);
        const nameMatch = code.match(/name:\s*['"]([^'"]+)['"]/);
        const versionMatch = code.match(/version:\s*['"]([^'"]+)['"]/);
        const iconMatch = code.match(/icon:\s*['"]([^'"]+)['"]/);
        
        return {
            id: idMatch ? idMatch[1] : 'unknown-' + Date.now(),
            name: nameMatch ? nameMatch[1] : '未命名工作栏',
            version: versionMatch ? versionMatch[1] : '1.0.0',
            icon: iconMatch ? iconMatch[1] : '📄'
        };
    }
    
    /**
     * 上线工作栏
     */
    async goOnline(columnId) {
        const column = this.warehouse.get(columnId);
        if (!column) {
            throw new Error('工作栏不存在');
        }
        
        try {
            // 执行工作栏代码
            eval(column.code);
            
            // 标记为已上线
            column.isOnline = true;
            this.onlineColumns.push(columnId);
            
            // 保存状态
            this._saveWarehouse();
            
            // 刷新显示
            this._refreshWarehouseList();
            this._refreshOnlineList();
            
            console.log(`[ColumnWarehouse] ✅ ${column.name} 已上线`);
            
        } catch (err) {
            console.error('[ColumnWarehouse] 上线失败:', err);
            throw new Error(`上线失败: ${err.message}`);
        }
    }
    
    /**
     * 下线工作栏
     */
    async goOffline(columnId) {
        const column = this.warehouse.get(columnId);
        if (!column) return;
        
        // 从ColumnRegistry中注销
        if (window.ColumnRegistry) {
            window.ColumnRegistry.unregister(columnId);
        }
        
        // 标记为未上线
        column.isOnline = false;
        this.onlineColumns = this.onlineColumns.filter(id => id !== columnId);
        
        // 保存状态
        this._saveWarehouse();
        
        // 刷新显示
        this._refreshWarehouseList();
        this._refreshOnlineList();
        
        console.log(`[ColumnWarehouse] ✅ ${column.name} 已下线`);
    }
    
    /**
     * 删除工作栏
     */
    deleteColumn(columnId) {
        const column = this.warehouse.get(columnId);
        if (!column) return;
        
        if (!confirm(`确定要删除 "${column.name}" 吗？此操作不可撤销！`)) {
            return;
        }
        
        // 如果已上线，先下线
        if (column.isOnline) {
            this.goOffline(columnId);
        }
        
        // 从仓库删除
        this.warehouse.delete(columnId);
        
        // 保存状态
        this._saveWarehouse();
        
        // 刷新显示
        this._refreshWarehouseList();
        
        console.log(`[ColumnWarehouse] ✅ ${column.name} 已删除`);
    }
    
    /**
     * 创建仓库UI
     */
    _createWarehouseUI() {
        const html = `
            <div id="column-warehouse-dialog" class="warehouse-dialog" style="display: none;">
                <div class="dialog-overlay" onclick="window.ColumnWarehouse.close()"></div>
                
                <div class="warehouse-panel">
                    <div class="warehouse-header">
                        <h2>📦 工作栏库房</h2>
                        <button class="close-btn" onclick="window.ColumnWarehouse.close()">✕</button>
                    </div>
                    
                    <div class="warehouse-body">
                        <div class="warehouse-split">
                            <!-- 左侧：库房列表 -->
                            <div class="warehouse-section">
                                <h3>📚 库房列表</h3>
                                <div id="warehouse-list" class="column-cards">
                                    <!-- 动态生成 -->
                                </div>
                                <button class="upload-btn" onclick="window.ColumnWarehouse.uploadColumn()">
                                    ⬆️ 上传工作栏
                                </button>
                            </div>
                            
                            <!-- 右侧：线上工作栏 -->
                            <div class="warehouse-section">
                                <h3>🚀 线上工作栏</h3>
                                <ul id="online-list" class="online-columns">
                                    <!-- 动态生成 -->
                                </ul>
                                <p class="hint">💡 提示：拖拽可调整顺序</p>
                            </div>
                        </div>
                    </div>
                    
                    <div class="warehouse-footer">
                        <button class="cancel-btn" onclick="window.ColumnWarehouse.close()">关闭</button>
                        <button class="apply-btn" onclick="window.ColumnWarehouse.applyChanges()">应用更改</button>
                    </div>
                </div>
            </div>
        `;
        
        document.body.insertAdjacentHTML('beforeend', html);
        this._addWarehouseStyles();
    }
    
    /**
     * 刷新库房列表
     */
    _refreshWarehouseList() {
        const container = document.getElementById('warehouse-list');
        container.innerHTML = '';
        
        this.warehouse.forEach((column, id) => {
            const card = document.createElement('div');
            card.className = 'column-card';
            card.innerHTML = `
                <div class="card-icon">${column.icon}</div>
                <div class="card-info">
                    <div class="card-name">${column.name}</div>
                    <div class="card-meta">
                        <span class="version">v${column.version}</span>
                        <span class="status ${column.isOnline ? 'online' : 'offline'}">
                            ${column.isOnline ? '✅ 已上线' : '未上线'}
                        </span>
                    </div>
                </div>
                <div class="card-actions">
                    ${!column.isOnline ? `
                        <button onclick="window.ColumnWarehouse.goOnline('${id}')" title="上线">
                            ⬆️ 上线
                        </button>
                    ` : `
                        <button onclick="window.ColumnWarehouse.goOffline('${id}')" title="下线">
                            ⬇️ 下线
                        </button>
                    `}
                    <button onclick="window.ColumnWarehouse.deleteColumn('${id}')" title="删除">
                        🗑️
                    </button>
                </div>
            `;
            container.appendChild(card);
        });
    }
    
    /**
     * 刷新线上列表
     */
    _refreshOnlineList() {
        const container = document.getElementById('online-list');
        container.innerHTML = '';
        
        this.onlineColumns.forEach((id, index) => {
            const column = this.warehouse.get(id);
            if (!column) return;
            
            const item = document.createElement('li');
            item.className = 'online-item';
            item.draggable = true;
            item.dataset.columnId = id;
            item.innerHTML = `
                <span class="order-number">${index + 1}</span>
                <span class="column-icon">${column.icon}</span>
                <span class="column-name">${column.name}</span>
                <button onclick="window.ColumnWarehouse.goOffline('${id}')">⬇️ 下线</button>
            `;
            container.appendChild(item);
        });
        
        // 初始化拖拽排序
        this._initOnlineListDrag();
    }
    
    /**
     * 初始化线上列表拖拽
     */
    _initOnlineListDrag() {
        // 类似之前的拖拽逻辑
        const list = document.getElementById('online-list');
        // ... 拖拽实现
    }
    
    /**
     * 保存到localStorage
     */
    _saveWarehouse() {
        const data = {
            warehouse: Array.from(this.warehouse.entries()),
            onlineColumns: this.onlineColumns
        };
        localStorage.setItem('column-warehouse', JSON.stringify(data));
    }
    
    /**
     * 从localStorage加载
     */
    _loadWarehouse() {
        const saved = localStorage.getItem('column-warehouse');
        if (saved) {
            const data = JSON.parse(saved);
            this.warehouse = new Map(data.warehouse);
            this.onlineColumns = data.onlineColumns || [];
        }
    }
    
    /**
     * 应用更改
     */
    applyChanges() {
        // 保存线上工作栏顺序到ColumnRegistry
        if (window.ColumnRegistry) {
            window.ColumnRegistry._applyColumnOrder(this.onlineColumns);
        }
        
        this.close();
    }
    
    // ... 样式方法
    _addWarehouseStyles() {
        // CSS样式（省略，类似之前的设置对话框）
    }
}

// 全局实例
window.ColumnWarehouse = new ColumnWarehouse();
```
