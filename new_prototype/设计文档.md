# PySide6 四栏项目管理器设计文档

## 1. 项目概述

本文档描述基于PySide6(Qt for Python)的四栏项目管理界面重构方案，旨在将原HTML/CSS/JS实现的原型升级为符合当前技术栈的桌面应用UI。

### 1.1 技术栈

- **UI框架**: PySide6 (Qt for Python 6.4+)
- **布局组件**: QSplitter, QTabWidget, QListWidget, QTextEdit
- **图表组件**: 
  - 脑图视图: 自定义QGraphicsView或PyQtGraph
  - 关系图: PyQtGraph或Qt+NetworkX
- **后端**: Python 3.9+ 与AutoGen 0.7.1
- **数据存储**: 本地JSON + Neo4j图数据库
- **向量数据库**: ChromaDB (AutoGen记忆组件)

### 1.2 功能需求

- 四栏布局: 项目列表、脑图视图、关系图和节点详情
- 拖拽调整面板大小
- 节点增删改查及层级关系维护
- 节点内容Markdown编辑和预览
- 关系图可视化与交互
- 标签管理与过滤
- 附件上传与管理

class Node:
    """脑图节点数据模型"""
    def __init__(self, node_id, title, parent_id=None):
        self.id = node_id
        self.title = title
        self.parent_id = parent_id
        self.children = []  # 子节点ID列表
        self.content = ""   # Markdown内容
        self.tags = []      # 标签列表
        self.attachments = []  # 附件列表
        self.created_at = datetime.now()
        self.updated_at = datetime.now()
        self.status = "active"  # 状态: active, archived, completed
        
    def to_dict(self):
        """转换为字典"""
        return {
            "id": self.id,
            "topic": self.title,  # 兼容jsMind格式
            "parent_id": self.parent_id,
            "children": self.children,
            "content": self.content,
            "tags": self.tags,
            "attachments": self.attachments,
            "created_at": self.created_at.isoformat(),
            "updated_at": self.updated_at.isoformat(),
            "status": self.status
        }
        
    @classmethod
    def from_dict(cls, data):
        """从字典创建节点"""
        node = cls(data["id"], data.get("topic", ""))
        node.parent_id = data.get("parent_id")
        node.children = data.get("children", [])
        node.content = data.get("content", "")
        node.tags = data.get("tags", [])
        node.attachments = data.get("attachments", [])
        
        # 处理日期时间字段
        if "created_at" in data:
            node.created_at = datetime.fromisoformat(data["created_at"])
        if "updated_at" in data:
            node.updated_at = datetime.fromisoformat(data["updated_at"])
            
        node.status = data.get("status", "active")
        return node

### 5.2 数据管理类

```python
class MindData:
    """脑图数据管理"""
    def __init__(self):
        self.nodes = {}  # id -> Node对象
        self.root_id = None
        
    def create_default(self):
        """创建默认脑图"""
        root = Node(str(uuid.uuid4()), "中心主题")
        self.nodes[root.id] = root
        self.root_id = root.id
        return root
        
    def find_node(self, node_id):
        """按ID查找节点"""
        return self.nodes.get(node_id)
        
    def find_parent_node(self, node_id):
        """查找父节点"""
        node = self.find_node(node_id)
        if node and node.parent_id:
            return self.find_node(node.parent_id)
        return None
        
    def add_child(self, parent_id, title):
        """添加子节点"""
        parent = self.find_node(parent_id)
        if not parent:
            return None
            
        node = Node(str(uuid.uuid4()), title, parent_id)
        self.nodes[node.id] = node
        parent.children.append(node.id)
        return node
        
    def update_content(self, node_id, content):
        """更新节点内容"""
        node = self.find_node(node_id)
        if node:
            node.content = content
            node.updated_at = datetime.now()
            return True
        return False
        
    def to_json(self):
        """转换为JSON格式"""
        data = {
            "meta": {
                "version": "1.0",
                "exported_at": datetime.now().isoformat()
            },
            "root_id": self.root_id,
            "nodes": {nid: node.to_dict() for nid, node in self.nodes.items()}
        }
        return data
        
    def load_from_json(self, data):
        """从JSON加载数据"""
        self.nodes = {}
        if "nodes" in data:
            for nid, node_data in data["nodes"].items():
                self.nodes[nid] = Node.from_dict(node_data)
                
        self.root_id = data.get("root_id")
        return True
```
