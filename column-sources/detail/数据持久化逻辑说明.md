# 详情页工作栏 - 数据持久化逻辑说明

## 🎯 **核心需求**

当用户在详情页编辑内容后，数据需要自动同步保存到数据底座：
- ✅ 旧节点 → 覆盖（UPDATE）
- ✅ 新节点 → 追加（INSERT）
- ✅ 自动判断是新节点还是旧节点

---

## 📊 **完整数据流**

```
用户编辑
    ↓
触发保存（自动/手动）
    ↓
获取 currentNode 数据
    ↓
查询数据底座：节点是否存在？
    ↓
    ├─ 存在 → 旧节点 → UPDATE（覆盖）
    │       ↓
    │   PUT /api/nodes/{nodeId}
    │       ↓
    │   更新成功 → 触发 node.saved 事件
    │
    └─ 不存在 → 新节点 → INSERT（追加）
            ↓
        POST /api/nodes
            ↓
        创建成功 → 触发 node.saved 事件
```

---

## 🔧 **技术实现**

### **1. 判断节点是否存在**

```javascript
/**
 * 检查节点是否存在于数据底座
 * @param {string} nodeId - 节点ID
 * @returns {Promise<boolean>} 是否存在
 */
async function checkNodeExists(nodeId) {
    try {
        // 方式1：通过API查询
        const response = await fetch(`/api/nodes/${nodeId}`);
        
        if (response.status === 200) {
            const data = await response.json();
            return data.found === true;
        }
        
        if (response.status === 404) {
            return false;  // 节点不存在
        }
        
        throw new Error(`查询失败: ${response.status}`);
        
    } catch (error) {
        console.error('[数据底座] 查询节点失败:', error);
        
        // 方式2：降级方案 - 从JSON文件加载
        try {
            const response = await fetch('test-data.json');
            const data = await response.json();
            const node = data.nodes.find(n => n.id === nodeId);
            return !!node;
        } catch (fallbackError) {
            console.error('[数据底座] 降级查询失败:', fallbackError);
            return false;
        }
    }
}
```

### **2. 保存节点数据（Upsert）**

```javascript
/**
 * 保存节点到数据底座（自动判断新增或更新）
 * @param {Object} nodeData - 完整节点数据
 * @returns {Promise<Object>} 保存结果
 */
async function saveNodeToDataBase(nodeData) {
    const nodeId = nodeData.id;
    
    console.log(`[数据底座] 准备保存节点: ${nodeId}`);
    
    // 步骤1：检查节点是否存在
    const exists = await checkNodeExists(nodeId);
    
    try {
        let response;
        let operation;
        
        if (exists) {
            // 步骤2a：旧节点 - 覆盖（UPDATE）
            console.log(`[数据底座] 节点已存在，执行UPDATE操作`);
            
            response = await fetch(`/api/nodes/${nodeId}`, {
                method: 'PUT',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify(nodeData)
            });
            
            operation = 'update';
            
        } else {
            // 步骤2b：新节点 - 追加（INSERT）
            console.log(`[数据底座] 节点不存在，执行INSERT操作`);
            
            response = await fetch('/api/nodes', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify(nodeData)
            });
            
            operation = 'insert';
        }
        
        // 步骤3：处理响应
        if (!response.ok) {
            throw new Error(`保存失败: ${response.status}`);
        }
        
        const result = await response.json();
        
        console.log(`[数据底座] 保存成功:`, {
            nodeId,
            operation,
            isNew: !exists
        });
        
        // 步骤4：触发事件
        window.dispatchEvent(new CustomEvent('node.saved', {
            detail: {
                nodeId: nodeId,
                data: nodeData,
                isNew: !exists,
                operation: operation,
                timestamp: Date.now()
            }
        }));
        
        return {
            success: true,
            operation: operation,
            isNew: !exists,
            data: result
        };
        
    } catch (error) {
        console.error('[数据底座] 保存失败:', error);
        
        // 降级方案：保存到localStorage
        try {
            localStorage.setItem(`node_${nodeId}`, JSON.stringify(nodeData));
            console.warn('[数据底座] 已降级保存到localStorage');
            
            return {
                success: true,
                operation: 'fallback',
                isNew: !exists,
                fallback: true
            };
        } catch (fallbackError) {
            console.error('[数据底座] 降级保存也失败:', fallbackError);
            throw error;
        }
    }
}
```

### **3. 自动保存（防抖）**

```javascript
/**
 * 自动保存管理器
 */
class AutoSaveManager {
    constructor() {
        this.saveTimer = null;
        this.saveInterval = 3000;  // 3秒防抖
        this.isDirty = false;  // 是否有未保存的更改
    }
    
    /**
     * 标记数据已修改
     */
    markDirty() {
        this.isDirty = true;
        this.scheduleSave();
    }
    
    /**
     * 调度保存（防抖）
     */
    scheduleSave() {
        // 清除之前的定时器
        if (this.saveTimer) {
            clearTimeout(this.saveTimer);
        }
        
        // 设置新的定时器
        this.saveTimer = setTimeout(async () => {
            if (this.isDirty) {
                await this.performSave();
            }
        }, this.saveInterval);
        
        console.log(`[自动保存] 已调度，${this.saveInterval}ms后保存`);
    }
    
    /**
     * 执行保存
     */
    async performSave() {
        if (!this.isDirty) return;
        
        try {
            console.log('[自动保存] 开始保存...');
            
            // 获取当前节点数据
            const currentNode = this.getCurrentNodeData();
            
            // 保存到数据底座
            const result = await saveNodeToDataBase(currentNode);
            
            if (result.success) {
                this.isDirty = false;
                console.log('[自动保存] 保存成功');
                
                // 更新UI状态
                this.updateSaveStatus('saved');
            }
            
        } catch (error) {
            console.error('[自动保存] 保存失败:', error);
            this.updateSaveStatus('error');
        }
    }
    
    /**
     * 获取当前节点数据
     */
    getCurrentNodeData() {
        return {
            id: document.getElementById('detail-title-input').dataset.nodeId,
            topic: document.getElementById('detail-title-input').value,
            content: document.getElementById('detail-content-editor').value,
            tags: this.getSelectedTags(),
            images: this.getPastedImages(),
            attachments: this.getAttachments(),
            updated_at: new Date().toISOString()
        };
    }
    
    /**
     * 更新保存状态UI
     */
    updateSaveStatus(status) {
        const statusEl = document.getElementById('save-status');
        if (!statusEl) return;
        
        const statusMap = {
            'dirty': '⚠️ 未保存',
            'saving': '⏳ 保存中...',
            'saved': '✅ 已保存',
            'error': '❌ 保存失败'
        };
        
        statusEl.textContent = statusMap[status] || '';
        statusEl.className = `save-status ${status}`;
    }
    
    // ... 其他辅助方法
}

// 初始化自动保存管理器
const autoSaveManager = new AutoSaveManager();

// 监听内容变化
document.getElementById('detail-content-editor').addEventListener('input', () => {
    autoSaveManager.markDirty();
});

// 监听标签变化
document.querySelectorAll('.tag-chip').forEach(chip => {
    chip.addEventListener('click', () => {
        autoSaveManager.markDirty();
    });
});
```


```
  "found": true,
```

## ⚙️ **配置选项**

```javascript
const persistenceConfig = {
    // 是否启用自动保存
    autoSave: true,
    
    // 自动保存间隔（毫秒）
    saveInterval: 3000,
    
    // 保存策略：upsert（自动判断） | update（仅更新） | insert（仅新增）
    saveStrategy: 'upsert',
    
    // 是否启用降级保存（localStorage）
    enableFallback: true,
    
    // API端点配置
    apiEndpoints: {
        check: '/api/nodes/{nodeId}',
        update: '/api/nodes/{nodeId}',
        insert: '/api/nodes',
        delete: '/api/nodes/{nodeId}'
    }
};
```
