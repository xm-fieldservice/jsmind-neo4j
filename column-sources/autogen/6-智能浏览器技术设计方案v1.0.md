## 🔧 核心组件设计

### 1. RouterAgent (路由决策)

**职责**: 分析问题,决定使用哪个Agent

```python
# backend/intelligent_browser/agents/router_agent.py
from autogen import AssistantAgent

class RouterAgent:
    """智能路由决策Agent"""
    
    def __init__(self, llm_config):
        self.agent = AssistantAgent(
            name="Router",
            system_message="""你是路由决策专家。
            分析用户问题,判断应该:
            1. local_rag - 查询本地文档
            2. web_search - 搜索网络
            3. hybrid - 两者结合
            """,
            llm_config=llm_config
        )
    
    def analyze_query(self, question: str) -> str:
        """分析问题类型"""
        # 使用LLM判断
        prompt = f"问题: {question}\n请判断查询类型(local_rag/web_search/hybrid)"
        response = self.agent.generate_reply(messages=[{"role": "user", "content": prompt}])
        return self.parse_decision(response)
```

### 2. LocalRAGAgent (本地检索)

**职责**: 使用LangChain检索本地文档

```python
# backend/intelligent_browser/agents/local_rag_agent.py
from autogen import AssistantAgent
from langchain.vectorstores import Chroma
from langchain.chains import RetrievalQA
from langchain.embeddings import OpenAIEmbeddings

class LocalRAGAgent:
    """本地文档RAG Agent"""
    
    def __init__(self, llm_config):
        # 初始化ChromaDB
        self.embeddings = OpenAIEmbeddings(
            openai_api_base="https://dashscope.aliyuncs.com/compatible-mode/v1",
            openai_api_key=os.getenv('DASHSCOPE_API_KEY'),
            model="text-embedding-v2"
        )
        
        self.vectorstore = Chroma(
            persist_directory="./data/chromadb/intelligent_browser",
            embedding_function=self.embeddings
        )
        
        # 创建Autogen Agent
        self.agent = AssistantAgent(
            name="LocalRAG",
            system_message="你是本地文档检索专家,使用RAG技术回答问题",
            llm_config=llm_config
        )
        
        # 注册LangChain工具
        self.register_tools()
    
    def register_tools(self):
        """注册LangChain RAG工具"""
        from langchain.tools import Tool
        
        def rag_search(query: str) -> str:
            """RAG检索工具"""
            qa_chain = RetrievalQA.from_chain_type(
                llm=self.get_llm(),
                retriever=self.vectorstore.as_retriever(search_kwargs={"k": 5})
            )
            return qa_chain.run(query)
        
        self.tools = [
            Tool(
                name="local_rag_search",
                func=rag_search,
                description="检索本地文档库"
            )
        ]
```

### 3. WebBrowserAgent (网络浏览)

**职责**: 使用Playwright浏览网页,提取信息

```python
# backend/intelligent_browser/agents/web_browser_agent.py
from autogen import AssistantAgent
from playwright.sync_api import sync_playwright
from langchain.tools import Tool

class WebBrowserAgent:
    """网络浏览Agent"""
    
    def __init__(self, llm_config):
        self.agent = AssistantAgent(
            name="WebBrowser",
            system_message="你是网络浏览专家,能够浏览网页并提取关键信息",
            llm_config=llm_config
        )
        
        self.register_tools()
    
    def browse_url(self, url: str) -> str:
        """浏览网页并提取内容"""
        with sync_playwright() as p:
            browser = p.chromium.launch(headless=True)
            page = browser.new_page()
            page.goto(url, wait_until='networkidle')
            
            # 提取主要内容
            content = page.evaluate("""() => {
                // 移除脚本和样式
                const scripts = document.querySelectorAll('script, style');
                scripts.forEach(s => s.remove());
                
                // 提取主要文本
                return document.body.innerText;
            }""")
            
            browser.close()
            return content
    
    def search_and_browse(self, query: str) -> str:
        """搜索并浏览多个结果"""
        # 使用搜索API获取URL列表
        urls = self.search_urls(query)
        
        # 浏览前3个结果
        results = []
        for url in urls[:3]:
            content = self.browse_url(url)
            results.append({
                'url': url,
                'content': content[:1000]  # 限制长度
            })
        
        # 使用LLM综合信息
        return self.synthesize_results(query, results)
    
    def register_tools(self):
        """注册网络浏览工具"""
        self.tools = [
            Tool(
                name="browse_url",
                func=self.browse_url,
                description="浏览指定URL并提取内容"
            ),
            Tool(
                name="search_and_browse",
                func=self.search_and_browse,
                description="搜索问题并浏览多个结果"
            )
        ]
```

## 🔌 API设计

### FastAPI接口

```python
# backend/intelligent_browser/main.py
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI(title="智能浏览器服务")

class QueryRequest(BaseModel):
    question: str
    mode: str = "auto"  # auto/local/web

@app.post("/api/intelligent-browser/ask")
async def ask_question(request: QueryRequest):
    """智能问答接口"""
    try:
        # 初始化Autogen系统
        coordinator = IntelligentBrowserCoordinator()
        
        # 执行查询
        result = coordinator.process_query(
            question=request.question,
            mode=request.mode
        )
        
        return {
            "success": True,
            "answer": result["answer"],
            "sources": result["sources"],
            "reasoning": result["reasoning"],
            "agent_used": result["agent"]
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
```
