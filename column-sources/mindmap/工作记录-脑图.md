class MindmapOperations {
    constructor(jm) {

    }

    // 统一的添加节点方法
    addNode(parentId, nodeId, topic, data = {}) {
        // 确保方向一致性
        if (!data.direction) {
            data.direction = 'right';
        }
        
        const parentNode = this.jm.get_node(parentId);
        if (parentNode) {

        }
        return null;

    // 统一的删除节点方法

        const node = this.jm.get_node(nodeId);
        if (node && !node.isroot) {
            this.jm.remove_node(nodeId);
            return true;
        }
        return false;
    }

    // 统一的移动节点方法
    moveNode(nodeId, targetParentId) {
        const node = this.jm.get_node(nodeId);
        const targetNode = this.jm.get_node(targetParentId);
        if (node && targetNode) {
            this.jm.move_node(nodeId, targetParentId);
            
            // 确保移动后方向一致
            node.direction = 'right';
            this.jm.view.reset();
            return true;
        }
        return false;
    }
