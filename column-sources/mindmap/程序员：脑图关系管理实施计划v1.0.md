#### 1.2 创建RelationManager类（2小时）
**文件**: `column-sources/mindmap/relation-manager.js`

```javascript
import { RELATION_TYPES, DEFAULT_RELATION_TYPE } from './relation-types.js';

/**
 * 关系管理器
 * 负责创建、管理节点间的关系
 */
export class RelationManager {
    constructor(jm) {
        this.jm = jm;
        
        // 复用现有架构组件
        this.storage = window.AutogenUnifiedStorage;
        this.eventBus = window.AutogenEventBus;
        this.errorHandler = window.ErrorHandler;
        
        console.log('[RelationManager] 初始化完成');
    }
    
    /**
     * 创建关系节点
     * @param {Object} sourceNode - 源节点
     * @param {Object} targetParent - 目标父节点
     * @param {String} relationType - 关系类型（默认RELATED_TO）
     * @returns {Object} 创建的关系节点
     */
    createRelationNode(sourceNode, targetParent, relationType = DEFAULT_RELATION_TYPE) {
        try {
            // 1. 验证参数
            if (!sourceNode || !targetParent) {
                throw new Error('源节点和目标父节点不能为空');
            }
            
            if (!RELATION_TYPES[relationType]) {
                console.warn(`[RelationManager] 未知关系类型: ${relationType}，使用默认类型`);
                relationType = DEFAULT_RELATION_TYPE;
            }
            
            // 2. 生成新节点ID
            const newNodeId = jsMind.util.uuid.newid();
            
            // 3. 构建关系数据
            const relationConfig = RELATION_TYPES[relationType];
            const relationData = this._buildRelationData(sourceNode, relationConfig);
            
            // 4. 添加关系符号到标题
            const newTopic = `${relationConfig.symbol} ${this._getOriginalTopic(sourceNode)}`;
            
            // 5. 使用jsMind API创建节点
            const newNode = this.jm.add_node(
                targetParent,
                newNodeId,
                newTopic,
                relationData
            );
            
            if (!newNode) {
                throw new Error('节点创建失败');
            }
            
            // 6. 设置节点颜色（使用jsMind内生API）
            this.jm.set_node_color(
                newNodeId,
                relationConfig.bgColor,
                relationConfig.fgColor
            );
            
            // 7. 递归复制子节点
            if (sourceNode.children && sourceNode.children.length > 0) {
                this._copyChildren(sourceNode, newNode, relationConfig);
            }
            
            // 8. 更新源节点计数器
            this._updateSourceNodeCounter(sourceNode, relationType);
            
            // 9. 触发事件（使用AutogenEventBus）
            this.eventBus.emit('mindmap.relation.created', {
                sourceNodeId: sourceNode.id,
                targetNodeId: newNodeId,
                relationType: relationType,
                relationSymbol: relationConfig.symbol,
                timestamp: new Date().toISOString()
            });
            
            // 10. 自动保存（使用AutogenUnifiedStorage）
            this._saveToStorage();
            
            console.log(`[RelationManager] 关系节点创建成功: ${newNodeId} (${relationType})`);
            return newNode;
            
        } catch (err) {
            // 使用ErrorHandler统一错误处理
            this.errorHandler.handle(err, {
                component: 'RelationManager',
                action: 'createRelationNode',
                context: {
                    sourceNodeId: sourceNode?.id,
                    targetParentId: targetParent?.id,
                    relationType
                }
            });
            return null;
        }
    }
    
    /**
     * 构建关系数据
     */
    _buildRelationData(sourceNode, relationConfig) {
        return {
            // 复制源节点的所有数据
            ...sourceNode.data,
            
            // 关系元数据
            isRelationNode: true,
            sourceNodeId: sourceNode.id,
            relationType: relationConfig.neo4jType,
            relationSymbol: relationConfig.symbol,
            relationStrength: relationConfig.strength,
            relationDirection: relationConfig.direction,
            createdAt: new Date().toISOString(),
            
            // 样式数据（jsMind内生支持）
            'background-color': relationConfig.bgColor,
            'foreground-color': relationConfig.fgColor
        };
    }
    
    /**
     * 获取节点的原始标题（去除计数器）
     */
    _getOriginalTopic(node) {
        if (node.data && node.data.original_topic) {
            return node.data.original_topic;
        }
        // 去除计数器标记 [复制×N]
        return node.topic.replace(/\s*\[复制×\d+\]\s*$/, '');
    }
    
    /**
     * 递归复制子节点
     */
    _copyChildren(sourceNode, targetNode, relationConfig) {
        sourceNode.children.forEach(child => {
            const childId = jsMind.util.uuid.newid();
            
            // 子节点继承关系标识，但不添加符号前缀
            const childData = {
                ...child.data,
                isRelationNode: true,
                sourceNodeId: child.data.sourceNodeId || sourceNode.id,
                relationType: relationConfig.neo4jType,
                'background-color': relationConfig.bgColor,
                'foreground-color': relationConfig.fgColor
            };
            
            const newChild = this.jm.add_node(
                targetNode,
                childId,
                child.topic,  // 子节点保持原标题
                childData
            );
            
            // 递归处理子节点的子节点
            if (child.children && child.children.length > 0) {
                this._copyChildren(child, newChild, relationConfig);
            }
        });
    }
    
    /**
     * 更新源节点计数器
     */
    _updateSourceNodeCounter(sourceNode, relationType) {
        try {
            // 初始化计数器
            if (!sourceNode.data.copy_count) {
                sourceNode.data.copy_count = 0;
                sourceNode.data.original_topic = this._getOriginalTopic(sourceNode);
            }
            
            // 初始化复制记录
            if (!sourceNode.data.copied_to) {
                sourceNode.data.copied_to = [];
            }
            
            // 增加计数
            sourceNode.data.copy_count++;
            
            // 记录复制信息
            sourceNode.data.copied_to.push({
                relationType: relationType,
                timestamp: new Date().toISOString()
            });
            
            // 更新标题显示计数器
            const newTopic = `${sourceNode.data.original_topic} [复制×${sourceNode.data.copy_count}]`;
            this.jm.update_node(sourceNode.id, newTopic);
            
            console.log(`[RelationManager] 源节点计数器更新: ${sourceNode.id} -> ${sourceNode.data.copy_count}`);
            
        } catch (err) {
            console.error('[RelationManager] 更新源节点计数器失败:', err);
        }
    }
    
    /**
     * 保存到统一存储
     */
    async _saveToStorage() {
        try {
            const mindmapData = this.jm.get_data();
            await this.storage.store('mindmap', 'current', mindmapData);
            console.log('[RelationManager] 数据已保存到AutogenUnifiedStorage');
        } catch (err) {
            console.error('[RelationManager] 保存失败:', err);
        }
    }
    
    /**
     * 获取所有关系节点
     */
    getAllRelationNodes() {
        const relations = [];
        const mindmapData = this.jm.get_data();
        
        const traverse = (node) => {
            if (node.data && node.data.isRelationNode) {
                relations.push({
                    id: node.id,
                    sourceNodeId: node.data.sourceNodeId,
                    relationType: node.data.relationType,
                    relationSymbol: node.data.relationSymbol,
                    topic: node.topic,
                    createdAt: node.data.createdAt
                });
            }
            
            if (node.children) {
                node.children.forEach(child => traverse(child));
            }
        };
        
        traverse(mindmapData.data);
        return relations;
    }
    
    /**
     * 统计关系信息
     */
    getRelationStats() {
        const relations = this.getAllRelationNodes();
        const stats = {
            total: relations.length,
            byType: {}
        };
        
        relations.forEach(rel => {
            if (!stats.byType[rel.relationType]) {
                stats.byType[rel.relationType] = 0;
            }
            stats.byType[rel.relationType]++;
        });
        
        return stats;
    }
}
```

```javascript
bindGlobalShortcuts() {
    // ... 现有代码 ...
    
    // 🔧 修改Ctrl+C - 复制节点
    document.addEventListener('keydown', (e) => {
        if ((e.ctrlKey || e.metaKey) && e.key === 'c') {
            const selectedNode = this.jm.get_selected_node();
            if (selectedNode && !selectedNode.isroot) {
                // 保存到剪贴板（复用现有NodeClipboard）
                if (this.clipboard) {
                    this.clipboard.copy(selectedNode);
                    console.log('[快捷键] 节点已复制:', selectedNode.topic);
                }
            }
        }
    });
    
    // 🆕 修改Ctrl+V - 粘贴为关系节点
    document.addEventListener('keydown', (e) => {
        if ((e.ctrlKey || e.metaKey) && e.key === 'v') {
            if (!this.clipboard || !this.clipboard.hasData()) {
                return;
            }
            
            const selectedNode = this.jm.get_selected_node();
            if (!selectedNode) {
                alert('请先选择目标父节点');
                return;
            }
            
            // 获取剪贴板数据
            const clipboardData = this.clipboard.clipboardData;
            
            // 使用RelationManager创建关系节点
            if (this.relationManager) {
                const newNode = this.relationManager.createRelationNode(
                    clipboardData,
                    selectedNode,
                    'RELATED_TO'  // 默认关联关系
                );
                
                if (newNode) {
                    this.jm.select_node(newNode.id);
                    console.log('[快捷键] 关系节点已创建:', newNode.topic);
                }
            }
            
            e.preventDefault();
        }
    });
}
```

```javascript
    const submenuTrigger = this.menu.querySelector('[data-action="relation-menu"]');

// 处理关系菜单动作
handleRelationAction(action) {
    const relationTypeMap = {
        'copy-related': 'RELATED_TO',
        'copy-strongly-related': 'STRONGLY_RELATED',
        'copy-influences': 'INFLUENCES',
        'copy-strongly-influences': 'STRONGLY_INFLUENCES',
        'copy-caused-by': 'CAUSED_BY',
        'copy-strongly-caused-by': 'STRONGLY_CAUSED_BY'
    };
    
    const relationType = relationTypeMap[action];
    if (!relationType) return;
    
    const selectedNode = this.jm.get_selected_node();
    if (!selectedNode) {
        alert('请先选择一个节点');
        return;
    }
    
    // 提示用户选择目标父节点
    this.showTargetSelectionHint();
    
    // 等待用户点击目标节点
    this.waitForTargetNode((targetNode) => {
        if (this.mindmapColumn.relationManager) {
            this.mindmapColumn.relationManager.createRelationNode(
                selectedNode,
                targetNode,
                relationType
            );
        }
    });
}
```

```javascript
// 在MindmapColumn类中添加拖拽处理
initDragCopy() {
    let dragSourceNode = null;
    let isDragging = false;
    let isRightButton = false;
    
    // 监听鼠标按下
    this.jm.view.e_panel.addEventListener('mousedown', (e) => {
        // 检测右键
        if (e.button === 2) {
            isRightButton = true;
            const node = this.jm.get_node_by_element(e.target);
            if (node && !node.isroot) {
                dragSourceNode = node;
                e.preventDefault();
            }
        }
    });
    
    // 监听鼠标移动
    this.jm.view.e_panel.addEventListener('mousemove', (e) => {
        if (isRightButton && dragSourceNode) {
            isDragging = true;
            // 显示拖拽提示
            this.showDragHint(e.clientX, e.clientY);
        }
    });
    
    // 监听鼠标释放
    this.jm.view.e_panel.addEventListener('mouseup', (e) => {
        if (isRightButton && isDragging && dragSourceNode) {
            // 获取目标节点
            const targetNode = this.jm.get_node_by_element(e.target);
            
            if (targetNode && targetNode.id !== dragSourceNode.id) {
                // 创建关系节点
                if (this.relationManager) {
                    this.relationManager.createRelationNode(
                        dragSourceNode,
                        targetNode,
                        'RELATED_TO'  // 右键拖拽默认为关联关系
                    );
                }
            }
            
            // 隐藏拖拽提示
            this.hideDragHint();
        }
        
        // 重置状态
        dragSourceNode = null;
        isDragging = false;
        isRightButton = false;
    });
    
    // 禁用右键菜单（避免冲突）
    this.jm.view.e_panel.addEventListener('contextmenu', (e) => {
        if (isDragging) {
            e.preventDefault();
        }
    });
}
```

```javascript
/**
 * 关系数据提取器
 * 用于导出关系数据到Neo4j
 */
export class RelationDataExtractor {
    constructor(jm) {
        this.jm = jm;
    }
    
    /**
     * 提取所有关系
     */
    extractRelations() {
        const relations = [];
        const mindmapData = this.jm.get_data();
        
        const traverse = (node) => {
            if (node.data && node.data.isRelationNode) {
                relations.push({
                    id: `rel_${node.id}`,
                    sourceNodeId: node.data.sourceNodeId,
                    targetNodeId: node.id,
                    type: node.data.relationType,
                    symbol: node.data.relationSymbol,
                    strength: node.data.relationStrength,
                    direction: node.data.relationDirection,
                    createdAt: node.data.createdAt,
                    properties: {
                        topic: node.topic,
                        bgColor: node.data['background-color'],
                        fgColor: node.data['foreground-color']
                    }
                });
            }
            
            if (node.children) {
                node.children.forEach(child => traverse(child));
            }
        };
        
        traverse(mindmapData.data);
        return relations;
    }
    
    /**
     * 转换为Neo4j Cypher
     */
    toNeo4jCypher(relations) {
        const statements = relations.map(rel => {
            return `
MATCH (source {id: '${rel.sourceNodeId}'})
MATCH (target {id: '${rel.targetNodeId}'})
CREATE (source)-[:${rel.type} {
    strength: '${rel.strength}',
    direction: '${rel.direction}',
    symbol: '${rel.symbol}',
    createdAt: '${rel.createdAt}'
}]->(target)`;
        });
        
        return statements.join('\n\n');
    }
    
    /**
     * 导出为JSON
     */
    toJSON() {
        const relations = this.extractRelations();
        return JSON.stringify(relations, null, 2);
    }
    
    /**
     * 下载为文件
     */
    downloadAsFile(filename = 'mindmap-relations.json') {
        const json = this.toJSON();
        const blob = new Blob([json], { type: 'application/json' });
        const url = URL.createObjectURL(blob);
        
        const a = document.createElement('a');
        a.href = url;
        a.download = filename;
        a.click();
        
        URL.revokeObjectURL(url);
    }
}
```
