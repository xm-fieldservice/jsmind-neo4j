#### 方向统一性保证

```javascript
// 1. 创建时直接指定方向
addChild() {
    const node = this.jm.add_node(selected, nodeId, '新节点', {
        direction: 'right'  // ✅ 创建时就指定方向
    });
}

// 2. 创建后立即设置方向
addBrother() {
    const node = this.jm.insert_node_after(selected, nodeId, '新节点');
    if (node) {
        node.direction = 'right';  // ✅ 创建后立即设置
        this.jm.view.reset();      // ✅ 刷新视图应用设置
    }
}

// 3. 自定义节点确保方向
addCustomNode(parent, topic, data = {}) {
    if (!data.direction) {
        data.direction = 'right';  // ✅ 默认右侧
    }
}
```

            this.autoSave();  // ✅ 自动保存
        }
    }
}

this.jm.add_event_listener((type, data) => {
