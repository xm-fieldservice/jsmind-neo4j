
# 节点操作统一化架构设计方案

## 🎯 设计目标

**核心目标**: 创建真正统一的节点操作架构
**质量标准**: 100%行为一致性
**验证标准**: 所有路径产生完全相同的用户体验

## 🏗️ 统一化架构设计

### 1. 统一核心方法 (核心引擎)

```javascript
// mindmap-operations.js - 统一核心引擎
class MindmapOperations {
    constructor(jm) {
        this.jm = jm;
    }

    /**
     * 统一节点创建核心引擎
     * 所有节点创建操作的唯一真实路径
     * @param {Object} params - 节点创建参数
     * @returns {Object} 创建的节点对象
     */
    _createNodeCore(params) {
        const {
            parent = null,
            topic = '新节点',
            direction = 'right',
            position = 'child', // 'child' | 'brother'
            selectedNode = null,
            customData = {}
        } = params;

        try {
            // 1. 生成节点ID
            const nodeId = this._generateNodeId();

            // 2. 创建时间戳
            const timestamp = this.getTimestamp();

            // 3. 基础数据准备
            const nodeData = {
                direction: direction,
                ...customData
            };

            // 4. 根据位置类型选择jsMind API
            let node;
            if (position === 'child') {
                // 创建子节点
                node = this.jm.add_node(parent, nodeId, topic, nodeData);
            } else if (position === 'brother') {
                // 创建兄弟节点
                if (selectedNode) {
                    node = this.jm.insert_node_after(selectedNode, nodeId, topic);
                    if (node) {
                        node.direction = direction; // 设置方向
                    }
                } else {
                    throw new Error('创建兄弟节点需要指定selectedNode');
                }
            }

            // 5. 统一后置处理 (关键！)
            if (node) {
                return this._postProcessNode(node, timestamp);
            }

            return null;
        } catch (error) {
            console.error('节点创建失败:', error);
            return null;
        }
    }

    /**
     * 统一节点后置处理
     * 所有节点创建后必须执行的标准化处理
     * @param {Object} node - 创建的节点
     * @param {string} timestamp - 时间戳
     * @returns {Object} 处理后的节点
     */
    _postProcessNode(node, timestamp) {
        // 1. 设置内容 (时间戳 + 空行)
        if (!node.data) node.data = {};
        node.data.content = timestamp + '\n\n';

        // 2. 选择节点
        this.jm.select_node(node.id);

        // 3. 进入编辑模式
        this.jm.begin_edit(node.id);

        // 4. 触发统一事件
        this._triggerNodeCreatedEvent(node);

        return node;
    }

    /**
     * 触发节点创建事件
     * 统一的事件通知机制
     * @param {Object} node - 创建的节点
     */
    _triggerNodeCreatedEvent(node) {
        // 触发统一事件，供其他模块监听
        if (this.jm && this.jm.eventBus) {
            this.jm.eventBus.emit('node:created', {
                node: node,
                timestamp: this.getTimestamp(),
                type: node._position || 'unknown' // 记录创建类型
            });
        }
    }

    /**
     * 生成唯一节点ID
     * @returns {string} 节点ID
     */
    _generateNodeId() {
        return 'node_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9);
    }

    /**
     * 获取格式化时间戳
     * @returns {string} 时间戳字符串
     */
    getTimestamp() {
        const now = new Date();
        const year = now.getFullYear();
        const month = String(now.getMonth() + 1).padStart(2, '0');
        const day = String(now.getDate()).padStart(2, '0');
        const hours = String(now.getHours()).padStart(2, '0');
        const minutes = String(now.getMinutes()).padStart(2, '0');
        const seconds = String(now.getSeconds()).padStart(2, '0');
        
        return `${year}-${month}-${day} ${hours}:${minutes}:${seconds}`;
    }

    // 2. 统一对外API方法 (封装层)

    /**
     * 添加子节点 - 统一API
     * @param {Object} parentNode - 父节点 (可选，默认使用选中节点)
     * @returns {Object} 创建的节点
     */
    addChild(parentNode = null) {
        const selected = parentNode || this.jm.get_selected_node();
        if (!selected) {
            console.warn('没有选中节点，无法添加子节点');
            return null;
        }

        return this._createNodeCore({
            parent: selected,
            topic: '新节点',
            direction: 'right',
            position: 'child'
        });
    }

    /**
     * 添加兄弟节点 - 统一API
     * @param {Object} selectedNode - 选中节点 (可选，默认使用当前选中节点)
     * @returns {Object} 创建的节点
     */
    addBrother(selectedNode = null) {
        const selected = selectedNode || this.jm.get_selected_node();
        if (!selected) {
            console.warn('没有选中节点，无法添加兄弟节点');
            return null;
        }

        return this._createNodeCore({
            selectedNode: selected,
            topic: '新节点',
            direction: 'right',
            position: 'brother'
        });
    }

    /**
     * 添加自定义节点 - 统一API
     * @param {Object} parentNode - 父节点
     * @param {string} topic - 节点主题
     * @param {Object} customData - 自定义数据
     * @returns {Object} 创建的节点
     */
    addCustomNode(parentNode, topic = '新节点', customData = {}) {
        if (!parentNode) {
            console.warn('必须指定父节点');
            return null;
        }

        return this._createNodeCore({
            parent: parentNode,
            topic: topic,
            direction: 'right',
            position: 'child',
            customData: customData
        });
    }
}
